class Solution {
    public ListNode addTwoNumbers(ListNode l1, ListNode l2) {
        ListNode a=new ListNode(0);
        ListNode cur=a;
        int b=0;
        while(l1!=null || l2!=null || b!=0){
            int total=b;
            if(l1!=null){
                total+=l1.val;
                l1=l1.next;
            }
            if(l2!=null){
                total+=l2.val;
                l2=l2.next;
            }
            b=total/10;
            cur.next=new ListNode(total%10);
            cur=cur.next;
        }
        return a.next;

    }
}
