import java.util.Scanner;
public class Solution {
    public static void main(String[] args) {
            Scanner sc=new Scanner(System.in);
            for(int i=0;i<3;i++){
                String s1 = sc.nextLine();
                String[] s2 = s1.split(" "); 
                String s3 = s2[0];
                int n = Integer.parseInt(s2[1]);
                System.out.printf("%-15s%03d%n", s3, n);
            }
    }
}
